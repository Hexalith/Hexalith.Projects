namespace Hexalith.Projects.Server.Folders;

using System.Diagnostics;
using System.Text.Json;
using Hexalith.EventStore.Contracts.Queries;
using Hexalith.EventStore.DomainService;
using Hexalith.Folders.Contracts;
using Hexalith.Projects.Contracts.Identifiers;
using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.Logging;

/// <summary>Uses verified delegated owner authorization on workload queries, preserving the existing direct-caller adapter.</summary>
/// <param name="client">The platform-owned delegated owner query client.</param>
/// <param name="accessor">The current server callback context.</param>
/// <param name="directCaller">The retained direct-caller adapter.</param>
/// <param name="logger">Optional closed-code diagnostics without protected request data.</param>
public sealed partial class DelegatedProjectFolderDirectory(Func<DelegatedOwnerQueryClient> client, IHttpContextAccessor accessor,
    IProjectFolderDirectory directCaller, ILogger<DelegatedProjectFolderDirectory>? logger = null) : IProjectFolderDirectory
{
    /// <inheritdoc/>
    public Task<ProjectFolderValidationResult> ValidateSetProjectFolderAsync(ProjectId projectId, string folderId, string correlationId, CancellationToken cancellationToken = default)
        => directCaller.ValidateSetProjectFolderAsync(projectId, folderId, correlationId, cancellationToken);

    /// <inheritdoc/>
    public async Task<ProjectFolderValidationResult> RefreshFolderReferenceAsync(ProjectId projectId, string folderId, string correlationId, CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(projectId);
        long started = Stopwatch.GetTimestamp();
        VerifiedQueryCaller? caller = VerifiedQueryCaller.Get(accessor.HttpContext);
        if (caller is null)
        {
            try
            {
                ProjectFolderValidationResult direct = await directCaller.RefreshFolderReferenceAsync(projectId, folderId, correlationId, cancellationToken).ConfigureAwait(false);
                if (direct.Outcome != ProjectFolderValidationOutcome.Accepted)
                {
                    Observe("direct-caller-non-accepted");
                }
                return direct;
            }
            catch (Exception)
            {
                Observe("direct-caller-exception");
                throw;
            }
        }
        try
        {
            var query = new SubmitQueryRequest(caller.Query.TenantId, "folders", folderId, FolderOwnerReadQuery.QueryType,
                Payload: JsonSerializer.SerializeToElement(new FolderOwnerReadQuery()));
            QueryResult result = await client().InvokeAsync(query, cancellationToken).ConfigureAwait(false);
            if (!result.Success)
            {
                Observe(ClassifyOwnerFailure(result.ErrorMessage));
                return new(result.ErrorMessage == "safe-denial" ? ProjectFolderValidationOutcome.Denied : ProjectFolderValidationOutcome.Unavailable, correlationId);
            }
            JsonElement payload = result.GetPayload();
            if (payload.ValueKind != JsonValueKind.Object)
            {
                Observe("invalid-owner-response");
                return new(ProjectFolderValidationOutcome.Unavailable, correlationId);
            }
            JsonProperty[] outcomes = payload.EnumerateObject().Where(static property => property.Name.Equals("outcome", StringComparison.OrdinalIgnoreCase)).ToArray();
            if (outcomes.Length != 1 || outcomes[0].Value.ValueKind != JsonValueKind.Number
                || !outcomes[0].Value.TryGetInt32(out int outcome) || !Enum.IsDefined((FolderOwnerReadOutcome)outcome))
            {
                Observe("invalid-owner-outcome");
                return new(ProjectFolderValidationOutcome.Unavailable, correlationId);
            }
            if ((FolderOwnerReadOutcome)outcome != FolderOwnerReadOutcome.Allowed)
            {
                Observe((FolderOwnerReadOutcome)outcome == FolderOwnerReadOutcome.Stale ? "owner-reported-stale" : "owner-reported-unavailable");
            }
            return new((FolderOwnerReadOutcome)outcome switch
            {
                FolderOwnerReadOutcome.Allowed => ProjectFolderValidationOutcome.Accepted,
                FolderOwnerReadOutcome.Stale => ProjectFolderValidationOutcome.Stale,
                _ => ProjectFolderValidationOutcome.Unavailable,
            }, correlationId);
        }
        catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
        {
            Observe("caller-operation-canceled");
            throw;
        }
        catch (Exception exception)
        {
            Observe(exception is OperationCanceledException ? "owner-operation-canceled"
                : exception is HttpRequestException ? "owner-transport-failed" : "owner-operation-failed");
            return new(ProjectFolderValidationOutcome.Unavailable, correlationId);
        }

        void Observe(string failureCode)
        {
            if (logger is not null)
            {
                LogOwnerFailure(logger, failureCode, Stopwatch.GetElapsedTime(started).TotalMilliseconds, cancellationToken.IsCancellationRequested);
            }
        }
    }

    /// <summary>Bounds remote failure text before it can reach runtime diagnostics.</summary>
    /// <param name="failure">The untrusted adapter-edge failure text.</param>
    /// <returns>A closed support code without remote text.</returns>
    internal static string ClassifyOwnerFailure(string? failure)
        => failure switch
        {
            "safe-denial" => "safe-denial",
            QueryAdapterFailureReason.QueryExecutionFailure => "owner-query-execution-failed",
            QueryAdapterFailureReason.ResponseMismatch => "owner-response-mismatch",
            QueryAdapterFailureReason.InvalidEnvelope => "invalid-owner-envelope",
            _ => "owner-query-failed",
        };

    [LoggerMessage(Level = LogLevel.Warning, Message = "Supported Folder owner read unavailable: FailureCode={FailureCode}; ElapsedMilliseconds={ElapsedMilliseconds}; CallerCanceled={CallerCanceled}.")]
    private static partial void LogOwnerFailure(ILogger logger, string failureCode, double elapsedMilliseconds, bool callerCanceled);
}
